4. JAVASCRIPT FILES AND ACTUAL FUNCTIONALITY
============================================
- index.js: waits for DOMContentLoaded; defines a navigation helper using querySelector, click listeners, and window.location.href for Login, Register, Start, and Join; adds a search feedback small element and input listener using trim and a template literal; appends a welcome paragraph using localStorage.getItem('loggedInUser') or developer fallback.
- login.js: selects login form fields; prepends a feedback paragraph; toggles password input type and button text; prevents submit; trims email; validates email with a regular expression and password length >= 6; writes loggedInUser; shows success; redirects to index.html after 900ms with setTimeout. It does not validate against users.
- signup.js: selects registration controls; creates/uses password strength small; prepends feedback; toggles password visibility; updates Weak/Medium/Strong on password input; prevents submit; trims username, lowercases email, validates username/email/password/confirmation; reads users JSON; scans users with for...of for duplicate email; pushes { username, email, password }; writes users JSON; resets form after success.
- BlogPage.js: defines six blog objects; filters by selected category and normalized search query; renders cards with filter/map/join and innerHTML; handles search input and category clicks/class state; routes Write a Blog, Login, and Register; renders at initialization. It does not read publishedBlogs.
- writeBlog.js: reads editor fields; creates feedback, dynamic character counter, and preview section; limits content to 5000 characters using slice; creates blog objects with title/category/content/author/tags/createdAt; validates required title/category/content; previews HTML; reads/pushes/writes drafts; reads/pushes/writes publishedBlogs; publishes and redirects after 900ms. The image input is not read, and the existing 0 words display is not updated.
- Developers.js: defines six developer objects; filters name/role/skills with trim, lowercase, join, includes; renders cards and initials with map/join; attaches profile click listeners with forEach; finds selected developer; stores selectedDeveloper JSON.
- DeveloperProfile.js: reads selectedDeveloper JSON with Ajay Kumar fallback; updates profile text and skills; builds connection:<name> key; reads/writes connection state and changes Connect to Connected; appends a backend-later message on Message click; if window.Chart exists, creates a canvas and initializes a bar chart with Projects 12, Blogs 24, Connections 186.
- Projects.js: defines six project objects; filters by category and search; sorts results by likes or views; renders cards and like buttons; finds project on like and increments in-memory likes once per rendered button; handles search, select changes, category clicks, active classes, and alert for Submit Project. Likes are not persisted.
- About.js: animates four number cards with setInterval, Math.max/ceil/min, and clearInterval; routes Join and dynamically added Get Started link to signup; toggles numbers seen class on scroll based on window.scrollY.

document.addEventListener('DOMContentLoaded', () => {
    const form = document.querySelector('.signup');
    const username = document.querySelector('#username');
    const email = document.querySelector('#email');
    const password = document.querySelector('#password');
    const confirmPassword = document.querySelector('#confirmPassword');
    const strength = document.querySelector('#password-strength') || document.createElement('small');
    if (!strength.parentElement) password.parentElement.appendChild(strength);
    const toggle = document.querySelector('#toggle-password');
    const message = document.createElement('p');
    form?.prepend(message);
    const showMessage = (text, success = false) => { message.textContent = text; message.style.color = success ? 'green' : 'crimson'; };
    toggle?.addEventListener('click', () => { const visible = password.type === 'text'; password.type = visible ? 'password' : 'text'; toggle.textContent = visible ? 'Show' : 'Hide'; });
    password?.addEventListener('input', () => { const length = password.value.length; strength.textContent = length < 6 ? 'Weak' : length < 10 ? 'Medium' : 'Strong'; });
    form?.addEventListener('submit', (event) => {
        event.preventDefault();
        const name = username.value.trim();
        const address = email.value.trim().toLowerCase();
        if (name.length < 3) showMessage('Username must contain at least 3 characters.');
        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(address)) showMessage('Enter a valid email address.');
        else if (password.value.length < 6) showMessage('Password must be at least 6 characters.');
        else if (password.value !== confirmPassword.value) showMessage('Passwords do not match.');
        else {
            const users = JSON.parse(localStorage.getItem('users') || '[]');
            let duplicate = false;
            for (const user of users) if (user.email === address) duplicate = true;
            if (duplicate) showMessage('This email is already registered.');
            else { users.push({ username: name, email: address, password: password.value }); localStorage.setItem('users', JSON.stringify(users)); showMessage('Account created. You can now log in.', true); form.reset(); }
        }
    });
});
